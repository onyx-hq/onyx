# CPU-only torch wheels. The default torch on Linux pulls ~1.5GB of NVIDIA
# CUDA wheels we'll never use (no GPU in this container). The `+cpu` local
# version qualifier forces pip to resolve from the PyTorch CPU index.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.12.0+cpu
torchvision==0.27.0+cpu

# Core sim-mode dependencies. Multi-arch (amd64 + arm64) all available as wheels.

httpx==0.28.1
# aiohttp powers the worker's tiny preview HTTP server (JPEG snapshots

# thumbnail via a proxy in oxy-cameras).
aiohttp==3.14.0
pydantic==2.10.3
psutil==6.1.0

# Real inference dependencies.

numpy==2.0.2

# Linear-assignment solver — required by Ultralytics' built-in trackers

# VLM client (compliance reports) + upsell-intent classifier. Optional at
# runtime: if ANTHROPIC_API_KEY is unset, compliance reports AND audio upsell
# detection are disabled and only camera_events flow.
anthropic==0.103.1

# On-device STT for the audio upsell pipeline (worker/upsell*.py). CTranslate2
# int8 Whisper — CPU-friendly, multi-arch wheels (amd64 + arm64); pulls
# ctranslate2 + onnxruntime (VAD) + av. Only imported when UPSELL_CAMERAS is
# set, but installed unconditionally so the small.en model can be baked into
# the image at build time (see Dockerfile) — first boot needs no network.
faster-whisper==1.2.1

# typo'd variable references rather than silently producing empty
# output. Built-in fallback (see prompts.py) is used when the
# backend ships an old config that doesn't include domain_pack.
Jinja2==3.1.6
